Q1.Write a program to count number of words in a string. 


#include <stdio.h>
void  main() {
    char str[100];
    int i, count = 1;

    printf("Enter a string: ");
    gets(str);

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] == ' ')
            count++;
    }

    printf("Number of words = %d", count);
} 

Q2.Write a program to print following series:
1 + 11+ 111+ 1111+ ..... + n times 1

#include <stdio.h>

void main()
{
    int n, i, a = 0;

    printf("Enter n: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        a = a * 10 + 1;
        printf("%d", a);
        if(i < n)
            printf(" + ");
    }
}
